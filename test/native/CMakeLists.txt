cmake_minimum_required(VERSION 3.22)
project(chatty_generation_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(NATIVE_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../android/app/src/main/cpp)
add_subdirectory(${NATIVE_ROOT}/llama_upstream llama)
add_executable(generation_output_test generation_output_test.cpp)
target_include_directories(generation_output_test PRIVATE ${NATIVE_ROOT} ${NATIVE_ROOT}/llama_upstream/src)
target_link_libraries(generation_output_test PRIVATE llama)
enable_testing()
add_test(NAME generation_output COMMAND generation_output_test)
